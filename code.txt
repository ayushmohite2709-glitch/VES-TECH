#include <SPI.h>
#include <MFRC522.h>
#include <WiFi.h>
#include <HTTPClient.h>

// ---- Wi-Fi config ----
const char* SSID = "BhaviCR";
const char* PASSWORD = "ayush270707";

// ---- API config ----
const char* VERIFY_URL = "http://192.168.0.105:3000/api/rfid/verify";

// ---- Pins ----
#define SS_PIN   5
#define RST_PIN  22
#define GREEN_LED 25
#define RED_LED   26
#define BUZZER    27

MFRC522 rfid(SS_PIN, RST_PIN);

void setup() {
  Serial.begin(115200);
  SPI.begin();
  rfid.PCD_Init();

  pinMode(GREEN_LED, OUTPUT);
  pinMode(RED_LED, OUTPUT);
  pinMode(BUZZER, OUTPUT);
  allOff();

  connectWiFi();

  Serial.println("Ready. Scan a card...");
}

void loop() {
  if (!rfid.PICC_IsNewCardPresent() || !rfid.PICC_ReadCardSerial()) {
    return;
  }

  String uid = getUidString();
  Serial.println("Scanned UID: " + uid);

  if (WiFi.status() != WL_CONNECTED) {
    Serial.println("WiFi disconnected, attempting reconnect...");
    connectWiFi();
  }

  if (WiFi.status() == WL_CONNECTED) {
    sendVerifyRequest(uid);
  } else {
    signalError(); // no wifi at all
  }

  rfid.PICC_HaltA();
  rfid.PCD_StopCrypto1();

  delay(1500); // debounce: avoid re-triggering on same tap
}

String getUidString() {
  String uid = "";
  for (byte i = 0; i < rfid.uid.size; i++) {
    if (rfid.uid.uidByte[i] < 0x10) uid += "0";
    uid += String(rfid.uid.uidByte[i], HEX);
    if (i < rfid.uid.size - 1) uid += ":";
  }
  uid.toUpperCase();
  return uid;
}

void connectWiFi() {
  WiFi.begin(SSID, PASSWORD);
  int attempts = 0;
  while (WiFi.status() != WL_CONNECTED && attempts < 20) {
    delay(500);
    Serial.print(".");
    attempts++;
  }
  if (WiFi.status() == WL_CONNECTED) {
    Serial.println("\nWiFi connected: " + WiFi.localIP().toString());
  } else {
    Serial.println("\nWiFi connection failed.");
  }
}

void sendVerifyRequest(String uid) {
  HTTPClient http;
  http.begin(VERIFY_URL);
  http.addHeader("Content-Type", "application/json");
  http.setTimeout(5000); // 5s timeout

  String payload = "{\"rfid_uid\":\"" + uid + "\"}";

  int retries = 1; // one retry on transient failure
  int httpCode = -1;
  String response = "";

  for (int attempt = 0; attempt <= retries; attempt++) {
    httpCode = http.POST(payload);
    if (httpCode == 200) {
      response = http.getString();
      break;
    }
    delay(300);
  }

  http.end();

  if (httpCode == 200) {
    handleResponse(response);
  } else {
    Serial.println("Request failed, httpCode: " + String(httpCode));
    signalError();
  }
}

void handleResponse(String response) {
  Serial.println("Response: " + response);

  // simple manual parse — no ArduinoJson dependency needed for MVP
  bool access = response.indexOf("\"access\":true") != -1;

  if (access) {
    signalAllowed();
  } else {
    signalDenied();
  }
}

// ---- Signal states ----

void allOff() {
  digitalWrite(GREEN_LED, LOW);
  digitalWrite(RED_LED, LOW);
  digitalWrite(BUZZER, LOW);
}

void signalAllowed() {
  allOff();
  digitalWrite(GREEN_LED, HIGH);
  digitalWrite(BUZZER, HIGH);
  delay(150);
  digitalWrite(BUZZER, LOW);
  delay(850);
  allOff();
}

void signalDenied() {
  allOff();
  digitalWrite(RED_LED, HIGH);
  for (int i = 0; i < 2; i++) {
    digitalWrite(BUZZER, HIGH);
    delay(120);
    digitalWrite(BUZZER, LOW);
    delay(120);
  }
  delay(600);
  allOff();
}

void signalError() {
  allOff();
  digitalWrite(BUZZER, HIGH); // one long continuous beep
  for (int i = 0; i < 3; i++) {
    digitalWrite(GREEN_LED, HIGH);
    digitalWrite(RED_LED, LOW);
    delay(200);
    digitalWrite(GREEN_LED, LOW);
    digitalWrite(RED_LED, HIGH);
    delay(200);
  }
  digitalWrite(BUZZER, LOW);
  allOff();
}
